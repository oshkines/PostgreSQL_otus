Домашнее задание 7: Журналы и контрольные точки
Выполнено на PostgreSQL 16 (Ubuntu 24.04, сервер Aeza)
=====================================================================

---------------------------------------------------------------------
1. НАСТРОЙКА КОНТРОЛЬНЫХ ТОЧЕК И ЛОГИРОВАНИЯ
---------------------------------------------------------------------

# Открываем конфигурационный файл PostgreSQL для редактирования
sudo nano /etc/postgresql/16/main/postgresql.conf

# В файле устанавливаем следующие параметры:
# checkpoint_timeout = 30s       (интервал между контрольными точками — 30 секунд)
# log_checkpoints = on           (включаем логирование контрольных точек)

# Перезапускаем кластер PostgreSQL, чтобы применить изменения
sudo pg_ctlcluster 16 main restart

---------------------------------------------------------------------
2. СНИМОК СТАТИСТИКИ «ДО» (перед запуском нагрузки)
---------------------------------------------------------------------

# Заходим в консоль psql (база testdb)
sudo -u postgres psql -d testdb

# Выполняем запрос для сбора начальных значений статистики WAL и контрольных точек
SELECT
    w.wal_records,
    w.wal_bytes,
    w.wal_buffers_full,
    b.checkpoints_timed,
    b.checkpoints_req
FROM pg_stat_wal AS w,
     pg_stat_bgwriter AS b;

# Записываем полученные значения:
# wal_bytes = 14 569 636 113
# checkpoints_timed = 2110
# checkpoints_req = 29

# Выходим из psql
\q

---------------------------------------------------------------------
3. ПРОГОН НАГРУЗКИ (10 МИНУТ) С СИНХРОННЫМ КОММИТОМ
---------------------------------------------------------------------

# Инициализируем базу testdb для pgbench (создаём тестовые таблицы, scale 10)
sudo -u postgres pgbench -i -s 10 testdb

# Запускаем нагрузку: 8 клиентов, 4 потока, длительность 600 секунд (10 минут)
sudo -u postgres pgbench -c 8 -j 4 -T 600 testdb

# Фиксируем результат TPS из вывода команды:
# tps = 690.101994 (without initial connection time)

---------------------------------------------------------------------
4. СНИМОК СТАТИСТИКИ «ПОСЛЕ» (после нагрузки)
---------------------------------------------------------------------

# Снова заходим в psql
sudo -u postgres psql -d testdb

# Повторяем тот же запрос статистики
SELECT
    w.wal_records,
    w.wal_bytes,
    w.wal_buffers_full,
    b.checkpoints_timed,
    b.checkpoints_req
FROM pg_stat_wal AS w,
     pg_stat_bgwriter AS b;

# Записываем полученные значения:
# wal_bytes = 17 388 581 022
# checkpoints_timed = 2146
# checkpoints_req = 29

# Выходим из psql
\q

---------------------------------------------------------------------
5. РАСЧЁТЫ (выполняются вручную или в отчёте)
---------------------------------------------------------------------

# Объем WAL за 10 минут:
# 17 388 581 022 - 14 569 636 113 = 2 818 944 909 байт ≈ 2688 МБ (2.6 ГБ)

# Количество плановых контрольных точек:
# 2146 - 2110 = 36

# Количество запрошенных контрольных точек:
# 29 - 29 = 0

# Средний объем WAL на одну контрольную точку:
# 2688 МБ / 36 ≈ 74.6 МБ

---------------------------------------------------------------------
6. СРАВНЕНИЕ РЕЖИМОВ КОММИТА: АСИНХРОННЫЙ (synchronous_commit = off)
---------------------------------------------------------------------

# Заходим в psql для изменения настройки
sudo -u postgres psql -d testdb

# Отключаем синхронный коммит (асинхронный режим)
ALTER SYSTEM SET synchronous_commit = off;

# Применяем настройку без перезапуска кластера
SELECT pg_reload_conf();

# Проверяем, что настройка применилась (должно быть off)
SHOW synchronous_commit;

# Выходим из psql
\q

# Запускаем нагрузку с теми же параметрами (8 клиентов, 4 потока, 600 секунд)
sudo -u postgres pgbench -c 8 -j 4 -T 600 testdb

# Фиксируем результат TPS из вывода команды:
# tps = 706.537835 (without initial connection time)

---------------------------------------------------------------------
7. ВОЗВРАТ В БЕЗОПАСНОЕ СОСТОЯНИЕ (synchronous_commit = on)
---------------------------------------------------------------------

# Заходим в psql
sudo -u postgres psql -d testdb

# Возвращаем синхронный коммит (важно для надёжности данных!)
ALTER SYSTEM SET synchronous_commit = on;

# Применяем настройку
SELECT pg_reload_conf();

# Проверяем, что настройка вернулась (должно быть on)
SHOW synchronous_commit;

# Выходим из psql
\q


Памятка к некоторым командам
pgbench -i -s 10 testdb — флаг -i означает инициализацию (создание таблиц), -s 10 — масштаб (количество строк), testdb — имя базы. Эта команда создаёт таблицы pgbench_accounts, pgbench_branches и др.

pgbench -c 8 -j 4 -T 600 testdb — -c 8 (8 клиентов), -j 4 (4 потока), -T 600 (длительность 600 секунд), testdb — база.

ALTER SYSTEM SET synchronous_commit = off; — изменяет настройку на уровне кластера (записывает в postgresql.auto.conf). Требует pg_reload_conf() для применения.

SELECT pg_reload_conf(); — перечитывает конфигурацию без перезапуска сервера.

SHOW synchronous_commit; — показывает текущее значение параметра.
